[0-9]+      { yylval = atoi(yytext); return NUMBER; }

"+"         { return PLUS; }
"-"         { return MINUS; }
"/"         { return DIV; }
"*"         { return MUL; }
"("         { return LPAREN; }
")"         { return RPAREN; }
.           { return yytext[0]; }

}


//clac.y

%{
    #include <stdio.h>
    #include <stdlib.h>
%}

%token NUMBER
%token PLUS MINUS DIV MUL LPAREN RPAREN NEWLINE

%left PLUS MINUS
%left MUL DIV
%right UMINUS

%%
input:
      input line
    |
    ;

line:
      expr NEWLINE { printf("Result: %d\n", $1); }
    | NEWLINE
    ;

expr:
      expr PLUS expr    { $$ = $1 + $3; }
    | expr MINUS expr   { $$ = $1 - $3; }
    | expr MUL expr     { $$ = $1 * $3; }
    | expr DIV expr     {
                            if ($3 == 0)
                                printf("Error: Cannot divide by zero!\n");
                            else
                                $$ = $1 / $3;
                        }
    | LPAREN expr RPAREN { $$ = $2; }
    | MINUS expr %prec UMINUS { $$ = -$2; }
    | NUMBER { $$ = $1; }
    ;
%%

int main()
{
    printf("Enter the expression (Ctrl+D to exit):\n");
    yyparse();
    return 0;
}

int yyerror(char *msg)
{
    printf("Syntax error: %s\n", msg);
    return 1;
}

//execute
yacc -d calc.y
lex calc.l
gcc y.tab.c lex.yy.c -ll -o calc
./calc
